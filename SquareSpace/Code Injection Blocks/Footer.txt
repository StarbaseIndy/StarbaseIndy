window.PodcastFeed = (function() {

  const RSS_URL =
    'https://api.riverside.com/hosting/DYO9Jfxk.rss';

  const API_URL =
    `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(RSS_URL)}`;

  let cachedFeed = null;

  function stripHtml(html) {
  
    const div = document.createElement('div');
  
    div.innerHTML = html;
  
    return div.textContent || div.innerText || '';
  }

  async function renderFeed(options) {
  
    const {
      containerId,
      limit = 10,
      showDescription = true,
      descriptionLength = 250
    } = options;
  
    const container =
      document.getElementById(containerId);
  
    if (!container) return;
  
    const items = await getFeed();
  
    container.innerHTML = items
      .slice(0, limit)
      .map(item => {
  
        const audioUrl =
          item.enclosure?.link || '';
  
        return `
          <div class="podcast-episode-card">
  
            <div class="podcast-title">${item.title}</div>
  
            <div class="podcast-date">
              ${new Date(
                item.pubDate
              ).toLocaleDateString()}
            </div>
  
            ${
              showDescription
                ? `
                  <p>
                    ${stripHtml(item.description).substring(
                      0,
                      descriptionLength
                    )}...
                  </p>
                `
                : ''
            }
  
            <audio controls>
              <source
                src="${audioUrl}"
                type="audio/mpeg">
            </audio>
  
            <p>
              <a href="${item.link}" target="_blank" rel="noopener noreferrer">
                Listen to Full Episode →
              </a>
            </p>
  
          </div>
        `;
      })
      .join('');
  }

  async function getFeed() {

    if (cachedFeed) return cachedFeed;

    try {
    
      const response = await fetch(API_URL);
      const data = await response.json();
    
      cachedFeed = data.items || [];
    
    } catch (err) {
    
      console.error('Podcast feed failed:', err);
    
      cachedFeed = [];
    }

    return cachedFeed;
  }

  async function renderEpisode(options) {

    const {
      containerId,
      episodeId,
      showDescription = true,
      showDate = true,
      descriptionLength = 250
    } = options;

    const container =
      document.getElementById(containerId);

    if (!container) return;

    const items = await getFeed();

    const episode = items.find(item =>
      item.guid &&
      item.guid.includes(episodeId)
    );

    if (!episode) {

      container.innerHTML =
        '<p>Episode not found.</p>';

      return;
    }

    const audioUrl =
      episode.enclosure?.link || '';

    container.innerHTML = `

      <div class="podcast-episode-card">

        <div class="podcast-title">${episode.title}</div>

        ${
          showDate
            ? `
              <div class="podcast-date">
                ${new Date(
                  episode.pubDate
                ).toLocaleDateString()}
              </div>
            `
            : ''
        }

        ${
          showDescription
            ? `
              <p>
                ${stripHtml(episode.description).substring(
                  0,
                  descriptionLength
                )}...
              </p>
            `
            : ''
        }

        <audio controls>
          <source
            src="${audioUrl}"
            type="audio/mpeg">
        </audio>

        <p>
          <a
            href="${episode.link}"
            target="_blank" rel="noopener noreferrer">
            Listen to Full Episode →
          </a>
        </p>

      </div>
    `;
  }

  async function renderLatest(options) {

    const {
      containerId
    } = options;

    const items = await getFeed();

    if (!items.length) return;

    renderEpisode({
      ...options,
      episodeId: items[0].guid
    });
  }

  return {
    renderFeed,
    renderEpisode,
    renderLatest
  };

})();
